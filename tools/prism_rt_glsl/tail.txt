/* skin, reset, mask: the objects' acceleration structures' inputs, made on
the GPU (xgpu_vulkan_rt.c's objects_record), with no descriptors (their
buffers by their device addresses, in the push constants).

skin: a captured draw's triangles each in an invocation (a workgroup is a
job's 64, the groups' table naming its job and its place in it): its three
corners' indices read (16- or 32-bit, a list, strip or fan), their vertices
read from the draw's own buffer (its position, float x y z; its two nodes,
bytes of three times the node's index, and the first's weight, a signed
16-bit fraction, where the layout has them) and skinned as the stock vertex
program skins them (the nodes' matrices, three rows each from c[60] of the
constants the draw was drawn with; none: the vertices are in the world
already, the mesh store's), written as the entry's triangle soup (nine
floats a triangle; a degenerate one, or one naming a vertex past the
draw's, written NaN: inactive in the structure). The entry's bounds and the
highest node byte its vertices name (plus one; a byte past the 44 nodes the
constants have, 253 in CE's own models for "none", is node 0's) are gathered in its info
(eight words: the bounds' least x y z and most x y z, as ordered words; the
node), a workgroup's at once.

reset: each job's entry's info made empty before skin.

mask: each instance of the frame's top level whose entry is an occluder
(the first-person weapon and hands among them) and whose bounds lie all
within 1 world unit of the eye left out (its mask 0), as CE never let the
first-person weapon cast a shadow. */

static const char skin_source[] =
	"#version 460\n"
	"#extension GL_EXT_buffer_reference : require\n"
	"#extension GL_EXT_buffer_reference_uvec2 : require\n"
	"layout(local_size_x = 64) in;\n"
	"layout(buffer_reference, std430, buffer_reference_align = 4) buffer Words { uint v[]; };\n"
	"layout(push_constant) uniform Push { uvec2 jobs; uvec2 groups; uvec2 info; uint count; uint pad; } push;\n"
	"shared uint low[3], high[3], most;\n"
	"\n"
	"uint ordered(float f)\n"
	"{\n"
	"    uint b = floatBitsToUint(f);\n"
	"    return (b & 0x80000000u) != 0u ? ~b : (b | 0x80000000u);\n"
	"}\n"
	"\n"
	"uint byte_at(Words w, uint at)\n"
	"{\n"
	"    return (w.v[at >> 2u] >> ((at & 3u) * 8u)) & 0xffu;\n"
	"}\n"
	"\n"
	"void main()\n"
	"{\n"
	"    Words groups = Words(push.groups);\n"
	"    uint job = groups.v[gl_WorkGroupID.x * 2u], within = groups.v[gl_WorkGroupID.x * 2u + 1u];\n"
	"    Words jobs = Words(push.jobs);\n"
	"    uint j = job * 24u;\n"
	"    Words vertices = Words(uvec2(jobs.v[j], jobs.v[j + 1u]));\n"
	"    Words indices = Words(uvec2(jobs.v[j + 2u], jobs.v[j + 3u]));\n"
	"    uvec2 constants_address = uvec2(jobs.v[j + 4u], jobs.v[j + 5u]);\n"
	"    Words constants = Words(constants_address);\n"
	"    Words soup = Words(uvec2(jobs.v[j + 6u], jobs.v[j + 7u]));\n"
	"    uint stride = jobs.v[j + 8u], position = jobs.v[j + 9u], nodes = jobs.v[j + 10u], weight = jobs.v[j + 11u];\n"
	"    uint index_size = jobs.v[j + 12u], primitive = jobs.v[j + 13u], triangles = jobs.v[j + 14u];\n"
	"    uint first = jobs.v[j + 15u], vertex_most = jobs.v[j + 17u], entry = jobs.v[j + 18u];\n"
	"    int index_add = int(jobs.v[j + 16u]);\n"
	"    bool skinned = any(notEqual(constants_address, uvec2(0u)));\n"
	"    if (gl_LocalInvocationIndex == 0u)\n"
	"    {\n"
	"        low[0] = low[1] = low[2] = 0xffffffffu;\n"
	"        high[0] = high[1] = high[2] = 0u;\n"
	"        most = 0u;\n"
	"    }\n"
	"    barrier();\n"
	"    uint t = within * 64u + gl_LocalInvocationIndex;\n"
	"    if (t < triangles)\n"
	"    {\n"
	"        uint c[3];\n"
	"        if (primitive == 0u) { c[0] = t * 3u; c[1] = t * 3u + 1u; c[2] = t * 3u + 2u; }\n"
	"        else if (primitive == 1u) { c[0] = t; c[1] = t + 1u; c[2] = t + 2u; }\n"
	"        else { c[0] = 0u; c[1] = t + 1u; c[2] = t + 2u; }\n"
	"        uint v[3];\n"
	"        bool ok = true;\n"
	"        for (int k = 0; k < 3; k++)\n"
	"        {\n"
	"            uint e = first + c[k], index;\n"
	"            if (index_size == 4u) index = indices.v[e];\n"
	"            else if (index_size == 2u) index = (indices.v[e >> 1u] >> ((e & 1u) * 16u)) & 0xffffu;\n"
	"            else index = c[k];\n"
	"            int at = int(index) + index_add;\n"
	"            ok = ok && at >= 0 && uint(at) < vertex_most;\n"
	"            v[k] = uint(max(at, 0));\n"
	"        }\n"
	"        ok = ok && v[0] != v[1] && v[1] != v[2] && v[0] != v[2];\n"
	"        vec3 p[3];\n"
	"        uint node_most = 0u;\n"
	"        for (int k = 0; k < 3 && ok; k++)\n"
	"        {\n"
	"            uint b = v[k] * stride + position;\n"
	"            vec3 q = uintBitsToFloat(uvec3(vertices.v[b >> 2u], vertices.v[(b >> 2u) + 1u], vertices.v[(b >> 2u) + 2u]));\n"
	"            if (skinned)\n"
	"            {\n"
	"                uint n0 = 0u, n1 = 0u;\n"
	"                float w = 1.0;\n"
	"                if (nodes != 0xffffffffu)\n"
	"                {\n"
	"                    n0 = byte_at(vertices, v[k] * stride + nodes);\n"
	"                    n1 = byte_at(vertices, v[k] * stride + nodes + 1u);\n"
	"                }\n"
	"                if (weight != 0xffffffffu)\n"
	"                {\n"
	"                    uint wb = v[k] * stride + weight;\n"
	"                    uint bits = (vertices.v[wb >> 2u] >> ((wb & 2u) * 8u)) & 0xffffu;\n"
	"                    w = float(int(bits << 16u) >> 16) / 32767.0;\n"
	"                }\n"
	"                node_most = max(node_most, max(n0 <= 129u ? n0 : 0u, n1 <= 129u ? n1 : 0u));\n"
	"                if (n0 > 129u) n0 = 0u;\n"
	"                if (n1 > 129u) n1 = 0u;\n"
	"                vec3 r;\n"
	"                for (uint axis = 0u; axis < 3u; axis++)\n"
	"                {\n"
	"                    uint a = (60u + n0 + axis) * 4u, z = (60u + n1 + axis) * 4u;\n"
	"                    vec4 ra = uintBitsToFloat(uvec4(constants.v[a], constants.v[a + 1u], constants.v[a + 2u], constants.v[a + 3u]));\n"
	"                    vec4 rb = uintBitsToFloat(uvec4(constants.v[z], constants.v[z + 1u], constants.v[z + 2u], constants.v[z + 3u]));\n"
	"                    r[axis] = w * dot(ra, vec4(q, 1.0)) + (1.0 - w) * dot(rb, vec4(q, 1.0));\n"
	"                }\n"
	"                q = r;\n"
	"            }\n"
	"            ok = ok && !any(isnan(q)) && !any(isinf(q));\n"
	"            p[k] = q;\n"
	"        }\n"
	"        uint o = t * 9u;\n"
	"        if (ok)\n"
	"        {\n"
	"            for (int k = 0; k < 3; k++)\n"
	"            {\n"
	"                soup.v[o + uint(k) * 3u] = floatBitsToUint(p[k].x);\n"
	"                soup.v[o + uint(k) * 3u + 1u] = floatBitsToUint(p[k].y);\n"
	"                soup.v[o + uint(k) * 3u + 2u] = floatBitsToUint(p[k].z);\n"
	"                for (uint axis = 0u; axis < 3u; axis++)\n"
	"                {\n"
	"                    atomicMin(low[axis], ordered(p[k][axis]));\n"
	"                    atomicMax(high[axis], ordered(p[k][axis]));\n"
	"                }\n"
	"            }\n"
	"            atomicMax(most, node_most + 1u);\n"
	"        }\n"
	"        else\n"
	"        {\n"
	"            for (uint k = 0u; k < 9u; k++)\n"
	"                soup.v[o + k] = 0x7fc00000u;\n"
	"        }\n"
	"    }\n"
	"    barrier();\n"
	"    if (gl_LocalInvocationIndex == 0u && most != 0u)\n"
	"    {\n"
	"        Words info = Words(push.info);\n"
	"        for (uint axis = 0u; axis < 3u; axis++)\n"
	"        {\n"
	"            atomicMin(info.v[entry * 8u + axis], low[axis]);\n"
	"            atomicMax(info.v[entry * 8u + 3u + axis], high[axis]);\n"
	"        }\n"
	"        atomicMax(info.v[entry * 8u + 6u], most);\n"
	"    }\n"
	"}\n";

static const char reset_source[] =
	"#version 460\n"
	"#extension GL_EXT_buffer_reference : require\n"
	"#extension GL_EXT_buffer_reference_uvec2 : require\n"
	"layout(local_size_x = 64) in;\n"
	"layout(buffer_reference, std430, buffer_reference_align = 4) buffer Words { uint v[]; };\n"
	"layout(push_constant) uniform Push { uvec2 jobs; uvec2 groups; uvec2 info; uint count; uint pad; } push;\n"
	"\n"
	"void main()\n"
	"{\n"
	"    uint job = gl_GlobalInvocationID.x;\n"
	"    if (job >= push.count)\n"
	"        return;\n"
	"    Words jobs = Words(push.jobs);\n"
	"    Words info = Words(push.info);\n"
	"    uint entry = jobs.v[job * 24u + 18u];\n"
	"    info.v[entry * 8u] = info.v[entry * 8u + 1u] = info.v[entry * 8u + 2u] = 0xffffffffu;\n"
	"    info.v[entry * 8u + 3u] = info.v[entry * 8u + 4u] = info.v[entry * 8u + 5u] = 0u;\n"
	"    info.v[entry * 8u + 6u] = 0u;\n"
	"}\n";

static const char mask_source[] =
	"#version 460\n"
	"#extension GL_EXT_buffer_reference : require\n"
	"#extension GL_EXT_buffer_reference_uvec2 : require\n"
	"layout(local_size_x = 64) in;\n"
	"layout(buffer_reference, std430, buffer_reference_align = 4) buffer Words { uint v[]; };\n"
	"layout(push_constant) uniform Push { uvec2 instances; uvec2 entries; uvec2 info; uint count; uint pad; vec4 eye; } push;\n"
	"\n"
	"float unordered(uint u)\n"
	"{\n"
	"    return uintBitsToFloat((u & 0x80000000u) != 0u ? (u & 0x7fffffffu) : ~u);\n"
	"}\n"
	"\n"
	"void main()\n"
	"{\n"
	"    uint i = gl_GlobalInvocationID.x;\n"
	"    if (i >= push.count)\n"
	"        return;\n"
	"    Words entries = Words(push.entries);\n"
	"    uint e = entries.v[i];\n"
	"    if ((e & 0x80000000u) == 0u)\n"
	"        return;\n"
	"    e &= 0x7fffffffu;\n"
	"    Words info = Words(push.info);\n"
	"    if (info.v[e * 8u + 6u] == 0u)\n"
	"        return;\n"
	"    float far = 0.0;\n"
	"    for (uint axis = 0u; axis < 3u; axis++)\n"
	"    {\n"
	"        float a = abs(unordered(info.v[e * 8u + axis]) - push.eye[axis]);\n"
	"        float b = abs(unordered(info.v[e * 8u + 3u + axis]) - push.eye[axis]);\n"
	"        far += max(a, b) * max(a, b);\n"
	"    }\n"
	"    if (far <= 1.0)\n"
	"    {\n"
	"        Words instances = Words(push.instances);\n"
	"        instances.v[i * 16u + 12u] &= 0x00ffffffu;\n"
	"    }\n"
	"}\n";

/* port: the structure's levels of detail (SCENE_STATIC's): the opaque
triangles' build input, each triangle as it is (V) where its level is the
one its surface is drawn at (or both), else folded to its first corner (no
area: never hit, its box a point inside the surface), before the structure
is refitted. Attributes' word 7: surface | level << 30 (XGPU_SCENE_LEVEL_*),
levels: a bit a surface, set where drawn far */
static const char levels_source[] =
	"#version 460\n"
	"#extension GL_EXT_buffer_reference : require\n"
	"#extension GL_EXT_buffer_reference_uvec2 : require\n"
	"layout(local_size_x = 64) in;\n"
	"layout(buffer_reference, std430, buffer_reference_align = 4) buffer Words { uint v[]; };\n"
	"layout(push_constant) uniform Push { uvec2 source; uvec2 target; uvec2 attributes; uvec2 levels; uint count;\n"
	"    uint words; } push;\n"
	"\n"
	"void main()\n"
	"{\n"
	"    uint t = gl_GlobalInvocationID.x;\n"
	"    if (t >= push.count)\n"
	"        return;\n"
	"    Words source = Words(push.source);\n"
	"    Words target = Words(push.target);\n"
	"    uint tag = Words(push.attributes).v[t * 8u + 7u];\n"
	"    uint level = tag >> 30u, surface = tag & 0x3fffffffu;\n"
	"    bool far = (surface >> 5u) < push.words && (Words(push.levels).v[surface >> 5u] & (1u << (surface & 31u))) != 0u;\n"
	"    bool drawn = level == 0u || (level == 1u && !far) || (level == 2u && far);\n"
	"    for (uint k = 0u; k < 9u; k++)\n"
	"        target.v[t * 9u + k] = source.v[t * 9u + (drawn ? k : k % 3u)];\n"
	"}\n";
